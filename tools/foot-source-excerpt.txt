function bodyMap() {
  const fig = FIGURES[CFG.figure] || FIGURES.silhouette;
  const ringR = fig.ring || 11;
  const [lxL, lxR] = fig.labelX;

  const spots = fig.spots.map(h => {
    const r = REG[h.id]; if (!r) return '';
    const left = h.side === 'l';
    const lx = left ? lxL : lxR;
    const leadFrom = left ? h.x - (ringR + 3) : h.x + (ringR + 3);
    const leadTo = left ? lxL + 6 : lxR - 6;
    const zones = (h.zones || []).map(z =>
      `<ellipse class="zone" cx="${z[0]}" cy="${z[1]}" rx="${z[2]}" ry="${z[3]}"/>`).join('');
    return `<g class="hot${h.dashed ? ' dashed' : ''}" role="link" tabindex="0" data-go="${h.id}"
              aria-label="${esc(r.name)} — examination and injection">
        ${zones}
        <line class="lead" x1="${leadFrom}" y1="${h.y}" x2="${leadTo}" y2="${h.ly}"/>
        <circle class="hit" cx="${h.x}" cy="${h.y}" r="${ringR + 10}" fill="#fff" fill-opacity="0"/>
        <circle class="ring" cx="${h.x}" cy="${h.y}" r="${ringR}"/>
        <circle class="dot" cx="${h.x}" cy="${h.y}" r="${ringR / 3}"/>
        <text class="lbl" x="${lx}" y="${h.ly}" text-anchor="${left ? 'end' : 'start'}">${esc(r.name)}</text>
        ${h.note ? `<text class="lbl" style="font-size:10.5px;font-weight:500;fill:var(--ink-3)"
              x="${lx}" y="${h.ly + 15}" text-anchor="${left ? 'end' : 'start'}">${h.note}</text>` : ''}
      </g>`;
  }).join('');

  return `<svg class="bodymap" viewBox="${fig.viewBox}" role="group" aria-label="Body map — choose a region">
      ${fig.art()}${spots}</svg>`;
}

let figGroups = [];

function isDose(items) {
  const joined = items.map(i => i.text).join(' ').toLowerCase();
  return items.length <= 6 && /needle|\d+\s*(mg|ml)|volume/.test(joined) &&
    items.every(i => i.text.length < 90 && !i.sub);
}

function doseList(items) {
  return `<ul class="doses">${items.map(i => {
    const m = i.text.match(/^(green|blue|orange|grey|gray|cream)\b/i);
    const total = /^total volume/i.test(i.text);
    const sw = m ? `<span class="swatch" style="background:${NEEDLE[m[1].toLowerCase()]}"></span>` : '';
    return `<li class="${total ? 'total' : ''}">${sw}${esc(i.text)}</li>`;
  }).join('')}</ul>`;
}

function figures(items) {
  const gi = figGroups.length;
  figGroups.push(items);
  const n = items.length;
  const cls = n === 1 ? 'n1' : n === 2 ? 'n2' : n === 3 ? 'n3' : n === 4 ? 'n4' : 'nmany';
  return `<div class="figs ${cls}">${items.map((f, i) => `
    <figure>
      <button class="figbtn" data-fig="${gi}" data-i="${i}" aria-label="Enlarge: ${esc(f.caption || 'clinical image')}">
        <img src="${IMG[f.src] || ''}" alt="${esc(f.caption || '')}" loading="lazy">
        <span class="zoom"><svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="#fff" stroke-width="2.4"
          stroke-linecap="round"><circle cx="11" cy="11" r="7"/><path d="M20 20l-3.6-3.6M11 8.4v5.2M8.4 11h5.2"/></svg></span>
      </button>
      ${f.caption ? `<figcaption>${f.n ? `<b>${f.n}</b>` : ''}${esc(f.caption)}</figcaption>` : ''}
    </figure>`).join('')}</div>`;
}

function tableBlock(b) {
  const cols = Math.max(...b.rows.map(r => r.length));
  if (cols === 2) {
    const cap = b.header && !/^(element|step|region|target|phase|presentation|movement|condition)$/i.test(b.header[0].trim())
      ? null : b.header;
    return `<dl class="spec">${b.rows.map(r =>
      `<div class="row"><dt>${esc(r[0])}</dt><dd>${esc(r[1] || '')}</dd></div>`).join('')}</dl>`;
  }
  return `<div class="tablewrap"><table>
    ${b.header ? `<thead><tr>${b.header.map(h => `<th>${esc(h)}</th>`).join('')}</tr></thead>` : ''}
    <tbody>${b.rows.map(r => `<tr>${r.map(c => `<td>${esc(c)}</td>`).join('')}</tr>`).join('')}</tbody>
  </table></div>`;
}
